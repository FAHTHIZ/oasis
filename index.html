async function extensionUpdate(name,scope,source,oldCommit){
  if(!agentToken||extensionBusy)return;
  extensionSetBusy(true);
  try{
    extensionStatus('Reviewing the latest pinned revision before updating '+name+'…');
    var data=await agentApi('/api/extensions/inspect',{method:'POST',body:JSON.stringify({url:source})});
    extensionRenderInspection(data,source);
    if(data.treeTruncated||data.canInstall===false){extensionStatus('Update blocked: repository tree is incomplete or has no supported Markdown prompts.',true);return}
    if((data.repository.commitSha||'').toLowerCase()===String(oldCommit||'').toLowerCase()){
      extensionStatus(name+' is already at the latest inspected commit.');
      return;
    }
    if(!window.confirm('Update '+name+' ('+scope+') from '+String(oldCommit||'').slice(0,12)+' to '+data.repository.commitSha.slice(0,12)+'?\\n\\nOasis verifies installed file hashes first and retains the previous revision for rollback. Edited files block the update. No scripts or MCP servers are enabled.'))return;
    extensionStatus('Verifying the installed files, staging the new revision, and keeping the previous version…');
    var result=await agentApi('/api/extensions/update',{method:'POST',body:JSON.stringify({url:source,scope:scope,expectedCommit:data.repository.commitSha})});
    if(result.alreadyCurrent){extensionStatus('Already current; no files changed.');return}
    await extensionRefreshInstalled();
    var backupText=(result.backups||[]).map(function(backup){return backup.scope+': '+backup.commitSha.slice(0,12)}).join(', ');
    extensionStatus('Updated '+name+' to '+result.commitSha.slice(0,12)+'. Rollback available for '+backupText+'.');
    notify('Extension updated with verified rollback history.');
  }catch(e){extensionStatus(e.message,true)}
  finally{extensionSetBusy(false)}
}
